function replaceRegex(text, regex, replacement, label) {
  if (text.includes(replacement)) return text;
  if (!regex.test(text)) throw new Error(`SOURCE DRIFT: ${label}`);
  return text.replace(regex, replacement);
}

function ensureIncludes(text, needle, label) {
  if (!text.includes(needle)) throw new Error(`SOURCE DRIFT after transform: ${label}`);
}
